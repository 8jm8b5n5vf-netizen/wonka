cmake_minimum_required(VERSION 3.22)
project(WONKA VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(WONKA_PLUGIN "Build JUCE plugin" ON)
enable_testing()
add_executable(volta_tests tests/DspTests.cpp Source/Engine.cpp)
target_include_directories(volta_tests PRIVATE Source)
add_test(NAME engine COMMAND volta_tests)
if(WONKA_PLUGIN)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
  set(formats VST3)
  if(APPLE)
    list(APPEND formats AU)
  endif()
  juce_add_plugin(WONKA COMPANY_NAME "Wonka Audio" BUNDLE_ID "audio.wonka.effect" PLUGIN_MANUFACTURER_CODE Wnka PLUGIN_CODE Wnk1 FORMATS ${formats} PRODUCT_NAME "WONKA" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(WONKA PRIVATE Source/Plugin.cpp Source/Editor.cpp Source/AudioIO.cpp Source/Engine.cpp)
  target_include_directories(WONKA PRIVATE Source)
  target_compile_definitions(WONKA PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(WONKA PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endif()
if(WONKA_PLUGIN)
  add_executable(volta_integration tests/PluginTests.cpp)
  target_include_directories(volta_integration PRIVATE Source "${juce_SOURCE_DIR}/modules")
  target_compile_definitions(volta_integration PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(volta_integration PRIVATE WONKA)
  add_test(NAME integration COMMAND volta_integration)
endif()
